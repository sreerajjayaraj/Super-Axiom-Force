cmake_minimum_required(VERSION 3.28)
project(SuperAxiomForce LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
set(FETCHCONTENT_SOURCE_DIR_ZYDIS "${CMAKE_CURRENT_SOURCE_DIR}/deps/zydis" CACHE PATH "" FORCE)
set(ZYAN_ZYCORE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/deps/zydis/dependencies/zycore" CACHE PATH "" FORCE)
set(ZYDIS_ZYCORE_PATH "${ZYAN_ZYCORE_PATH}" CACHE PATH "" FORCE)
add_subdirectory(deps/safetyhook)
add_library(SuperAxiomForce SHARED src/SuperAxiomForce/SuperAxiomForce.cpp
    targeting/src/UnifiedTargeting.cpp deps/safetyhook/allocator_win.cpp)
target_compile_definitions(SuperAxiomForce PRIVATE NOMINMAX SAF_GAME_RELEASE)
target_compile_options(SuperAxiomForce PRIVATE /W4 /WX /EHsc /guard:cf)
target_link_options(SuperAxiomForce PRIVATE /guard:cf /Brepro)
target_link_libraries(SuperAxiomForce PRIVATE safetyhook psapi)
set_target_properties(SuperAxiomForce PROPERTIES PREFIX "" SUFFIX ".asi")
